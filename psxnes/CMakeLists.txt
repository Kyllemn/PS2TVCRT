cmake_minimum_required(VERSION 3.21)
project(psxnes LANGUAGES C ASM)
option(PSXNES_DYNAREC "recompilador 6502->MIPS" ON)
set(SRC psx/main.c psx/video.c psx/sound.c psx/save.c core/nes.c core/cpu.c core/apu.c)
if(PSXNES_DYNAREC)
  list(APPEND SRC psx/dynarec.c psx/dyn.S)
endif()
psn00bsdk_add_executable(psxnes GPREL ${SRC})
target_compile_definitions(psxnes PRIVATE NES_GPU)
if(PSXNES_DYNAREC)
  target_compile_definitions(psxnes PRIVATE DYNAREC)
endif()
target_compile_options(psxnes PRIVATE -O2)
